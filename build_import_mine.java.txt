package com.simulationtown.building;

import com.buildlib.structure.StructureNbtParser;
import net.neoforged.fml.loading.FMLPaths;

import javax.swing.JFileChooser;
import javax.swing.filechooser.FileNameExtensionFilter;
import java.awt.GraphicsEnvironment;
import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.util.List;
import java.util.stream.Stream;

public class build_import {

    private static final String ROOT_DIR = "simulation_town_buliding";
    private static final String INCOMING = "_incoming";

    /** 方块名 -> 分类目录 */
    private static final String[][] CATEGORY_BLOCKS = {
            {"simulation_town:residence_block", "residence"},
            {"simulation_town:agriculture_block", "agriculture"},
            {"simulation_town:industry_block", "industry"},
            {"simulation_town:special_block", "special"},
    };

    public String daoru() {
        // 开发环境（Gradle run）是 headless：JFileChooser 会抛 HeadlessException 并崩游戏
        if (GraphicsEnvironment.isHeadless()) {
            return daoruFromIncoming();
        }

        JFileChooser chooser = new JFileChooser();
        chooser.setFileFilter(new FileNameExtensionFilter("NBT 文件", "nbt"));
        if (chooser.showOpenDialog(null) != JFileChooser.APPROVE_OPTION) {
            return "已取消选择";
        }
        File file = chooser.getSelectedFile();
        if (file == null || !file.isFile()) {
            return "没有选中文件";
        }
        return importOne(file.toPath());
    }

    /** 扫描 _incoming 文件夹，把里面的 .nbt 全部导入。 */
    public String daoruFromIncoming() {
        Path incoming = FMLPaths.GAMEDIR.get().resolve(ROOT_DIR).resolve(INCOMING);
        try {
            Files.createDirectories(incoming);
            List<Path> files;
            try (Stream<Path> stream = Files.list(incoming)) {
                files = stream.filter(p -> p.toString().toLowerCase().endsWith(".nbt")).toList();
            }
            if (files.isEmpty()) {
                return "没有待导入文件。把 .nbt 放进 " + incoming;
            }
            int ok = 0;
            String lastError = null;
            for (Path p : files) {
                String r = importOne(p);
                if (r.startsWith("已导入")) {
                    ok++;
                } else {
                    lastError = r;
                }
            }
            if (ok == 0) {
                return "一个都没导入成：" + lastError;
            }
            return "已导入 " + ok + " 个建筑" + (lastError == null ? "" : "（有失败：" + lastError + "）");
        } catch (Exception e) {
            return "导入失败：" + e.getMessage();
        }
    }

    private String importOne(Path file) {
        String fileName = file.getFileName().toString();
        List<String> names;
        try {
            names = StructureNbtParser.parsePaletteNames(file.toString());
        } catch (Exception e) {
            return "读不出文件：" + fileName + "（" + e.getMessage() + "）";
        }
        if (names == null || names.isEmpty()) {
            return "读不出 palette：" + fileName;
        }

        String category = null;
        for (String n : names) {
            if (n == null) continue;
            for (String[] entry : CATEGORY_BLOCKS) {
                if (entry[0].equalsIgnoreCase(n)) {
                    category = entry[1];
                    break;
                }
            }
            if (category != null) break;
        }
        if (category == null) {
            return "错误的建筑格式：" + fileName + "（palette 里没有建筑方块）";
        }

        Path dir = FMLPaths.GAMEDIR.get().resolve(ROOT_DIR).resolve(category);
        Path target = uniqueTarget(dir, fileName);
        if (target == null) {
            return "同名文件太多了：" + fileName;
        }
        try {
            Files.createDirectories(dir);
            Files.copy(file, target, StandardCopyOption.REPLACE_EXISTING);
        } catch (Exception e) {
            return "复制失败：" + e.getMessage();
        }

        build_block.refresh();   // 让列表下一次扫描立刻重读
        String saved = target.getFileName().toString();
        return "已导入 " + category + "：" + saved.substring(0, saved.length() - 4);
    }

    /** 同名不覆盖，改成「名字 (2).nbt」。 */
    private static Path uniqueTarget(Path dir, String fileName) {
        Path target = dir.resolve(fileName);
        if (!Files.exists(target)) return target;

        String base = fileName;
        String ext = "";
        int dot = fileName.lastIndexOf('.');
        if (dot > 0) {
            base = fileName.substring(0, dot);
            ext = fileName.substring(dot);
        }
        for (int i = 2; i < 1000; i++) {
            target = dir.resolve(base + " (" + i + ")" + ext);
            if (!Files.exists(target)) return target;
        }
        return null;
    }
}
